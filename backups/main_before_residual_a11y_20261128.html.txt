    const CHINESE_GROUP_UNITS = [

    function spellUnderTenThousand(value, leadingOne = false) {
      const digits = PINYIN_DIGITS;
      let out = '';
      const thousands = Math.floor(value / 1000);
      if (thousands) out += digits[thousands] + '千';
      let rest = value % 1000;
      const hundreds = Math.floor(rest / 100);
      if (hundreds) {
        out += digits[hundreds] + '百';
        rest %= 100;
        if (rest === 0) return out;
        if (rest < 10) out += '〇';
      } else if (thousands && rest > 0) {
        out += '〇';
      }
      const tens = Math.floor(rest / 10);
      if (tens) {
        // Standalone 10-19 is 十/十一, but after a higher unit it is 一十/一十一.
        if (tens !== 1 || out || leadingOne) out += digits[tens];
        out += '十';
      }
      const ones = rest % 10;
      if (ones) out += digits[ones];
      return out || '〇';
    }

    // Group the digit string instead of using Number so large values are not
    // rounded before they are spelled out.
    function spellChineseInteger(value) {
      const digits = String(value).replace(/^0+(?=\d)/, '');
      if (digits === '0') return '〇';
      // No named unit exists above 极 (10^48). Digit-by-digit reading is
      // faithful; omitting the unit would silently corrupt the value.
      if (Math.ceil(digits.length / 4) > CHINESE_GROUP_UNITS.length) {
        return Array.from(digits, d => PINYIN_DIGITS[Number(d)]).join('');
      }
      const groups = [];
      for (let end = digits.length; end > 0; end -= 4) {
        groups.push(Number(digits.slice(Math.max(0, end - 4), end)));
      }
      let out = '';
      let previousIndex = groups.length;
      for (let i = groups.length - 1; i >= 0; i--) {
        const group = groups[i];
        if (!group) continue;
        // RBNF inserts 〇 before a small final group and after skipped groups.
        // A single 〇 covers both cases at once.
        // RBNF only inserts the gap 〇 immediately before the lowest group.
        const hasGap = i === 0 && previousIndex - i > 1;
        const smallFinal = i === 0 && Boolean(out) && group < 1000;
        if (out && (hasGap || smallFinal)) out += '〇';
        const fullTens = i === 0 && Boolean(out) && group >= 10 && group < 20;
        out += spellUnderTenThousand(group, fullTens) + (CHINESE_GROUP_UNITS[i] || '');
        previousIndex = i;
      }
      return out || '〇';
    }

      const unsigned = source.replace(/^[-+]\s*/, '');
      const parts = unsigned.split('.');
      const integerDigits = parts[0] || '0';
      const integerText = spellChineseInteger(integerDigits);
      const fraction = parts.length > 1 ? parts[1] : '';
      const fractionHasDigits = Boolean(fraction) && !/^0+$/.test(fraction);
      if (negative) {
        // RBNF drops negative fractions and renders -0 as 〇.
        if (/^0+$/.test(integerDigits)) return fractionHasDigits ? '负' : '〇';
        return '负' + integerText;
      }
      if (fractionHasDigits) {
        return integerText + '点' + Array.from(fraction, d => PINYIN_DIGITS[Number(d)]).join('');
      }
      return integerText;
    }

    // Numeric years, temperatures and percentages are resolved here rather
    // than with /\d+(?:\.\d+)?\s*unit/ regexes: those backtrack across the
    // whole digit run whenever the unit is absent, which is quadratic.
    function matchNumericUnit(text, index) {
      let j = index;
      while (j < text.length && /\s/.test(text[j])) j++;
      if (j >= text.length) return null;
      if (text[j] === '年') return { type: 'year', end: j + 1 };
      if (text[j] === '%' || text[j] === '％') return { type: 'percent', end: j + 1 };
      if (text[j] === '℃') return { type: 'temp', end: j + 1 };
      if (text[j] === '°') {
        let k = j + 1;
        while (k < text.length && /\s/.test(text[k])) k++;
        if (k < text.length && text[k] === 'C') return { type: 'temp', end: k + 1 };
      }
      return null;
    }

    function replaceNumericRuns(text) {
      let out = '';
      let i = 0;
      while (i < text.length) {
        if (!/\d/.test(text[i]) && text[i] !== '-' && text[i] !== '−') {
          const plainStart = i;
          while (i < text.length && !/\d/.test(text[i]) && text[i] !== '-' && text[i] !== '−') i++;
          out += text.slice(plainStart, i);
          continue;
        }
        const start = i;
        // Preserve earlier signs verbatim and let the last one drive the
        // negative reading: --1% -> -百分之负一 (same as the old regexes).
        while (i < text.length && (text[i] === '-' || text[i] === '−')) i++;
        const signRun = text.slice(start, i);
        const sign = signRun ? '-' : '';
        const prefixSigns = signRun.slice(0, -1);
        const multipleSigns = signRun.length > 1;
        if (signRun) {
          // A sign separated from its digits by spaces still applies to the
          // following number ("- 5%" -> 百分之负五). Keep the gap only when
          // no digit follows it.
          while (i < text.length && /\s/.test(text[i])) i++;
        }
        if (i >= text.length || !/\d/.test(text[i])) {
          out += text.slice(start, i);
          continue;
        }
        let integer = '';
        while (i < text.length && /\d/.test(text[i])) integer += text[i++];
        let fraction = '';
        if (i < text.length && text[i] === '.') {
          const dot = i;
          i += 1;
          let fractionDigits = '';
          while (i < text.length && /\d/.test(text[i])) fractionDigits += text[i++];
          if (fractionDigits) fraction = '.' + fractionDigits;
          else i = dot;
        }
        const unit = matchNumericUnit(text, i);
        let spelled;
        if (unit && unit.type === 'year' && !fraction && integer.length === 4) {
          spelled = signRun + Array.from(integer, d => PINYIN_DIGITS[Number(d)]).join('') + '年';
        } else {
          if (unit && unit.type === 'temp') {
            // The sign is expressed by 零下, so do not let spellChineseNumber
            // emit 负 as well.
            spelled = prefixSigns + (sign ? '零下' : '') + spellChineseNumber(integer + fraction) + '度';
          } else if (unit && unit.type === 'percent') {
            spelled = prefixSigns + '百分之' + spellChineseNumber(sign + integer + fraction);
          } else {
            // Old behavior for bare numbers: repeated signs stay verbatim and
            // only a single sign contributes the negative reading.
            spelled = (multipleSigns ? signRun : '') +
              spellChineseNumber((multipleSigns ? '' : sign) + integer + fraction);
            if (unit && unit.type === 'year') spelled += '年';
          }
        }
        out += spelled;
        if (unit) i = unit.end;
      }
      return out;
    }

    // Chinese-run year/percent units are scanned once. A regex like
    // /[〇零...]+年/g backtracks the whole run at every start position when
    // the unit is absent, which is quadratic on long text.
    function expandChineseUnitRuns(source) {
      const unitChar = /[〇零一二三四五六七八九十百千万亿两点]/;
      const yearChar = /[〇零一二三四五六七八九]/;
      let out = '';
      let i = 0;
      while (i < source.length) {
        if (!unitChar.test(source[i])) {
          const plainStart = i;
          while (i < source.length && !unitChar.test(source[i])) i++;
          out += source.slice(plainStart, i);
          continue;
        }
        const start = i;
        while (i < source.length && unitChar.test(source[i])) i++;
        const run = source.slice(start, i);
        let j = i;
        while (j < source.length && /\s/.test(source[j])) j++;
        if (j < source.length && source[j] === '年') {
          // The old regex matched the trailing 年-eligible suffix of a run,
          // e.g. 十三 -> 十三年 and 一百零一 -> 一百〇一年.
          let suffixStart = run.length;
          while (suffixStart > 0 && yearChar.test(run[suffixStart - 1])) suffixStart--;
          if (suffixStart < run.length) {
            out += run.slice(0, suffixStart) + run.slice(suffixStart).replace(/零/g, '〇') + '年';
            i = j + 1;
          } else {
            out += run;
          }
        } else if (j < source.length && (source[j] === '%' || source[j] === '％')) {
          out += '百分之' + run;
          i = j + 1;
        } else {
          out += run;
        }
      }
      return out;
    }

    function expandChineseNumbers(text) {
      let source = String(text).replace(/[“”"]/g, '');
      // Upstream RBNF/Python \d accepts full-width digits. Normalize them
      // before any scanner runs so ７７% and ２０２６年 are expanded too.
      source = source.replace(/[０-９]/g, ch => ch.normalize('NFKC'));
      // Thousands separators are punctuation, not digits: "1,000" is 一千.
      source = source.replace(/(\d)[,，](?=\d{3}(?!\d))/g, '$1');
      // Written years use digit-by-digit 〇, never 零 (二零二六年 -> 二〇二六年).
      // Numeric units are handled by replaceNumericRuns() in the same pass as
      // bare numbers, so no regex can partially match a 65+ digit run.
      source = expandChineseUnitRuns(source);
      return replaceNumericRuns(source);
    }

    function initCanvas(saveState = true) {
      cancelAnalysis();
      keyDrawing = false;
      if (saveState) saveUndoState();
      ctx.globalCompositeOperation = 'source-over';
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';
      undoBtn.disabled = undoStack.length === 0;
    }
    initCanvas(false);
    undoBtn.disabled = true;

    function saveUndoState() {
      try {
        const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
        undoStack.push(imageData);
        if (undoStack.length > MAX_UNDO) undoStack.shift();
        undoBtn.disabled = false;
      } catch (e) {
        console.warn('Undo state save failed:', e);
      }
    }

    function undo() {
      cancelAnalysis();
      // A pointer stroke must not resume from the old coordinate after its
      // saved state has been popped.
      drawing = false;
      if (keyDrawing) setKeyPen(false);
      if (undoStack.length === 0) {
        showMessage('元に戻す操作がありません。');
        undoBtn.disabled = true;
        return;
      }
      const imageData = undoStack.pop();
      ctx.putImageData(imageData, 0, 0);
      undoBtn.disabled = undoStack.length === 0;
    }

      const y = (clientY - rect.top) * (canvas.height / rect.height);

    }

    function startDrawing(e) {
      e.preventDefault();

      drawLine(last.x, last.y, p.x, p.y);
      last = p;

      ctx.strokeStyle = tool === 'pen' ? penColor : 'rgba(0,0,0,1)';

    // ---- ローカル同梱アセットを優先し、無ければ CDN を使う ----
    async function resolveAsset(localPath, remoteUrl, { requireSri = true } = {}) {
      try {
        const res = await fetch(localPath, { method: 'HEAD', cache: 'no-store' });
        if (res.ok) {
          // The vendored files are byte-identical to the CDN copies, so the
          // same hash can verify them at runtime without a second download.
          // This closes the gap where only CDN fallbacks were SRI-checked.
          return { url: localPath, local: true, integrity: CDN_SRI[remoteUrl] || '' };
        }
      } catch (e) { /* local assets are optional */ }
      const integrity = CDN_SRI[remoteUrl] || '';
      // SRI 未定義の CDN を無検証で実行しない。音声モデルは
      // PIPER_ASSET_SRI で別途検証するためのみオプトアウトを許可する。
      if (requireSri && !integrity) {
        throw new Error(`CDN フォールバックの SRI が定義されていません: ${remoteUrl}`);
      }
      return { url: remoteUrl, local: false, integrity };
    }

    async function requireLocalAssets(paths) {
      const results = await Promise.all(paths.map(async (path) => {
        try {
          return (await fetch(path, { method: 'HEAD', cache: 'no-store' })).ok;
        } catch (e) {
          return false;
        }
      }));
      return results.every(Boolean);
    }

    // ---- 動的スクリプト読み込み ----
    // A failed <script> element must not poison later retries, and concurrent
    // callers should join one real in-flight load instead of any DOM node.
    const pendingScripts = new Map();
    function loadScript(src, integrity = '') {
      const existing = document.querySelector(`script[src="${src}"]`);
      if (existing && existing.dataset.loaded === 'true') {
        return Promise.resolve();
      }
      if (pendingScripts.has(src)) {
        return pendingScripts.get(src);
      }
      const promise = new Promise((resolve, reject) => {
        const script = document.createElement('script');
        script.src = src;
        if (integrity) {
          script.integrity = integrity;
          script.crossOrigin = 'anonymous';
        }
        script.onload = () => {
          script.dataset.loaded = 'true';
          pendingScripts.delete(src);
          resolve();
        };
        script.onerror = () => {
          script.remove();
          pendingScripts.delete(src);
          reject(new Error('Failed to load: ' + src));
        };
        document.head.appendChild(script);
      });
      pendingScripts.set(src, promise);
      return promise;
    }

    // ---- 空キャンバスチェック ----
    function isCanvasEmpty() {
      const pixels = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
      for (let i = 0; i < pixels.length; i += 4) {
        if (pixels[i+3] > 0 && (pixels[i] !== 255 || pixels[i+1] !== 255 || pixels[i+2] !== 255)) {
          return false;
        }
      }
      return true;
    }

    // ---- 画像前処理（OCR精度向上） ----
    function preprocessCanvas() {
      const w = canvas.width;
      const h = canvas.height;
      const imageData = ctx.getImageData(0, 0, w, h);
      const data = imageData.data;
      // Grayscale + Otsu threshold
      const gray = new Uint8Array(w * h);
      let min = 255, max = 0;
      for (let i = 0; i < w * h; i++) {
        const a = data[i*4+3] / 255;
        const r = data[i*4] * a + 255 * (1 - a);
        const g = data[i*4+1] * a + 255 * (1 - a);
        const b = data[i*4+2] * a + 255 * (1 - a);
        const v = Math.round(0.299*r + 0.587*g + 0.114*b);
        gray[i] = v;
        if (v < min) min = v;
        if (v > max) max = v;
      }
      // Contrast stretch
      const range = max - min || 1;
      for (let i = 0; i < w * h; i++) {
        gray[i] = Math.round((gray[i] - min) / range * 255);
      }
      // Otsu threshold
      const hist = new Array(256).fill(0);
      for (let i = 0; i < w * h; i++) hist[gray[i]]++;
      let total = w * h;
      let sum = 0;
      for (let i = 0; i < 256; i++) sum += i * hist[i];
      let sumB = 0, wB = 0, maxVar = 0, threshold = 128;
      for (let t = 0; t < 256; t++) {
        wB += hist[t];
        if (wB === 0) continue;
        const wF = total - wB;
        if (wF === 0) break;
        sumB += t * hist[t];
        const mB = sumB / wB;
        const mF = (sum - sumB) / wF;
        const between = wB * wF * (mB - mF) * (mB - mF);
        if (between > maxVar) {
          maxVar = between;
          threshold = t;
        }
      }
      // Apply threshold: dark text on white background
      const out = ctx.createImageData(w, h);
      for (let i = 0; i < w * h; i++) {
        const v = gray[i] > threshold ? 255 : 0;
        out.data[i*4] = v;
        out.data[i*4+1] = v;
        out.data[i*4+2] = v;
        out.data[i*4+3] = 255;
      }
      // Create temp canvas with processed image
      const tmpCanvas = document.createElement('canvas');
      tmpCanvas.width = w;
      tmpCanvas.height = h;
      tmpCanvas.getContext('2d').putImageData(out, 0, 0);
      return tmpCanvas.toDataURL('image/png');
    }

    function getApiUrl() {
      const raw = apiUrlInput.value.trim().replace(/\/$/, '');
      if (!raw) return '';
      try {
        const url = new URL(raw);
        if (url.protocol !== 'http:' && url.protocol !== 'https:') return '';
        if (url.username || url.password) return '';
        const host = url.hostname.toLowerCase();
        const loopback = host === '127.0.0.1' || host === 'localhost' || host === '::1' || host === '[::1]';
        if (!loopback) return '';
        return url.toString().replace(/\/$/, '');
      } catch (e) {
        return '';
      }
    }

    // ---- Tesseract.js OCR (高性能オフラインOCR) ----
    async function initTesseract({ manageProgress = true } = {}) {
      if (tesseractReady) return true;
      // Cancelling an analysis cannot abort createWorker(). Join the in-flight
      // initialization instead of starting a second worker from a second run.
      if (tesseractInitPromise) return tesseractInitPromise;
      const promise = (async () => {
        try {
          if (manageProgress) showProgress(true);
          setProgress(10);
          const tesseractAsset = await resolveAsset('vendor/tesseract/tesseract.min.js', TESSERACT_CDN);
          if (!(await requireLocalAssets(TESSERACT_LOCAL_ASSETS))) {
            throw new Error('OCRの二次アセットがローカルにありません。tools/fetch-assets.py を実行してください。');
          }
          if (!window.Tesseract) await loadScript(tesseractAsset.url, tesseractAsset.integrity);
          // The CDN main script has SRI, but Tesseract fetches these files itself.
          // Always point them at the verified local vendor copies.
          const localOptions = {
            workerPath: new URL('vendor/tesseract/worker.min.js', location.href).href,
            corePath: new URL('vendor/tesseract/core/', location.href).href,
            langPath: new URL('vendor/tesseract/lang/', location.href).href
          };
          tesseractOptions = Object.assign({
            logger: m => {
              if (m.status === 'recognizing text' && progressBar.style.display !== 'none') {
                setProgress(30 + Math.round(m.progress * 60));
              }
            }
          }, localOptions);
          setProgress(30);
          tesseractWorker = await Tesseract.createWorker('chi_sim', 1, tesseractOptions);
          setProgress(100);
          tesseractReady = true;
          if (manageProgress) showProgress(false);
          return true;
        } catch (err) {
          console.error('Tesseract init error:', err && err.message ? err.message : err);
          if (manageProgress) showProgress(false);
          showMessage('OCRエンジンの読み込みに失敗しました。' + (err && err.message ? ' ' + err.message : ''), 0);
          return false;
        }
      })();
      tesseractInitPromise = promise;
      try {
        return await promise;
      } finally {
        if (tesseractInitPromise === promise) tesseractInitPromise = null;
      }
    }

    async function runOCR(imageDataUrl, { manageProgress = true } = {}) {
      const ok = await initTesseract({ manageProgress });
      if (!ok || !tesseractWorker) {
        throw new Error('Tesseract初期化失敗');
      }
      // Try multiple PSM modes for best results
      const modes = [
        { psm: 10, label: 'single char' },   // Single character
        { psm: 8, label: 'single word' },     // Single word
        { psm: 13, label: 'raw line' },       // Raw line (no OSD)
      ];
      let bestResult = '';
      let bestConfidence = -1; // confidence 0 の候補も採用するため
      for (const mode of modes) {
        try {
          await tesseractWorker.setParameters({ tessedit_pageseg_mode: mode.psm });
          const { data } = await tesseractWorker.recognize(imageDataUrl);
          const text = (data.text || '').trim();
          const conf = data.confidence || 0;
          const match = text.match(/[\u4e00-\u9fff]/);
          if (match && conf > bestConfidence) {
            bestResult = match[0];
            bestConfidence = conf;
          }
          if (bestConfidence > 80) break; // Good enough
        } catch (e) {
          console.warn(`PSM ${mode.psm} failed:`, e);
        }
      }
      return bestResult;
    }

        addVoiceOption(voiceSelect, '', 'ブラウザ音声なし（Piperは利用可）');

      if (!values.includes(target)) target = browserVoiceDefault();
      if (values.includes(target)) voiceSelect.value = target;
      updateSelectedVoice();
      updateTtsControls();
    }

    const PLAYBACK_LABELS = new Map([

      [playPinyinBtn, playExampleBtn].forEach(b => {
        b.classList.remove('playing');
        b.textContent = '▶ 再生';
        b.setAttribute('aria-label', (PLAYBACK_LABELS.get(b) || '音声') + 'を再生');
      });

    }

    function beginPlaybackUI(btn, token) {
      if (token !== playbackToken) return false;
      resetPlaybackUI();
      activeButton = btn;
      btn.classList.add('playing');
      btn.textContent = '⏹ 停止';
      btn.setAttribute('aria-label', (PLAYBACK_LABELS.get(btn) || '音声') + 'を停止');
      return true;
    }

    function endPlaybackUI(token) {
      if (token === playbackToken) resetPlaybackUI();
    }

    function restoreFallbackVoice() {
      if (!fallbackVoiceRestore) return;
      const { restore, temporary } = fallbackVoiceRestore;
      fallbackVoiceRestore = null;
      // A different selection made during the fallback belongs to the user.
      if (voiceSelect.value !== temporary) return;
      voiceSelect.value = restore;
      updateSelectedVoice();
      updateTtsControls();
    }

    function stopSpeaking() {
      // Stop and replacement playback both end the temporary fallback.
      restoreFallbackVoice();
      playbackToken++;
      // A remote Piper download can stall; Stop must release it immediately.
      // Clear the in-flight record synchronously too: a replacement playback in
      // the same task must not join a promise whose signal was just aborted.
      piperLoading = null;
      piperFetchControllers.forEach(controller => controller.abort());
      piperFetchControllers.clear();
      if (activeAudio) {
        const audio = activeAudio;
        activeAudio = null;
        audio.onended = null;
        audio.onerror = null;
        try { audio.pause(); audio.currentTime = 0; } catch (e) { /* ignore */ }
      }
      if (activeAudioUrl) {
        URL.revokeObjectURL(activeAudioUrl);
        activeAudioUrl = null;
      }
      if (activeUtterance) {
        activeUtterance.onend = null;
        activeUtterance.onerror = null;
        activeUtterance = null;
      }
      if (window.speechSynthesis) window.speechSynthesis.cancel();
      resetPlaybackUI();
      setTtsStatus('');
    }

    function speakWithBrowser(text, btn, token, statusText = 'ブラウザ音声で再生中', onFinish = null) {
      const finish = () => {
        if (token === playbackToken && typeof onFinish === 'function') onFinish();
      };
      if (!window.speechSynthesis || typeof window.SpeechSynthesisUtterance === 'undefined') {
        showMessage('このブラウザは音声合成に対応していません。');
        endPlaybackUI(token);
        setTtsStatus('');
        finish();
        return;
      }
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'zh-CN';
      if (selectedVoice) utterance.voice = selectedVoice;
      utterance.rate = ttsRate;
      utterance.pitch = ttsPitch;
      utterance.onend = () => {
        if (activeUtterance === utterance) activeUtterance = null;
        endPlaybackUI(token);
        setTtsStatus('');
        finish();
      };
      utterance.onerror = (e) => {
        if (activeUtterance === utterance) activeUtterance = null;
        if (e && (e.error === 'canceled' || e.error === 'interrupted')) {
          endPlaybackUI(token);
          setTtsStatus('');
          finish();
          return;
        }
        console.error('Speech error:', e);
        endPlaybackUI(token);
        setTtsStatus('');
        showMessage('音声の再生に失敗しました。');
        finish();
      };
      if (!beginPlaybackUI(btn, token)) return;
      activeUtterance = utterance;
      window.speechSynthesis.cancel();
      window.speechSynthesis.speak(utterance);
      setTtsStatus(statusText);
    }

    async function loadPiperRuntime() {
      if (window.ort && window.pinyinPro && typeof window.pinyinPro.pinyin === 'function') return;
      setTtsStatus('ニューラル音声のエンジンを読み込み中...');
      const ortAsset = await resolveAsset('vendor/ort/ort.min.js', ORT_BASE + 'ort.min.js');
      if (!(await requireLocalAssets(ORT_LOCAL_ASSETS))) {
        throw new Error('PiperのWASMアセットがローカルにありません。tools/fetch-assets.py を実行してください。');
      }
      await loadScript(ortAsset.url, ortAsset.integrity);
      const pinyinAsset = await resolveAsset('vendor/pinyin/pinyin-pro.js', PINYIN_PRO_BASE + 'index.js');
      await loadScript(pinyinAsset.url, pinyinAsset.integrity);
      if (!window.ort || !window.pinyinPro || typeof window.pinyinPro.pinyin !== 'function') {
        throw new Error('Piper runtime load failed');
      }
      // ORT loads .mjs/.wasm internally; use the verified local copies even
      // when the main script came from the SRI-protected CDN.
      window.ort.env.wasm.wasmPaths = new URL('vendor/ort/', location.href).href;
      window.ort.env.wasm.numThreads = 1;
      window.ort.env.wasm.proxy = false;
    }

    async function verifyBlobIntegrity(blob, integrity) {
      if (!integrity) return;
      if (!window.crypto || !crypto.subtle) {
        throw new Error('整合性検証に必要な Web Crypto API が利用できません。');
      }
      const expected = /^sha256-([A-Za-z0-9+/]+={0,2})$/.exec(integrity);
      if (!expected) throw new Error('不正な SRI 形式です: ' + integrity);
      const digest = await crypto.subtle.digest('SHA-256', await blob.arrayBuffer());
      const actual = 'sha256-' + btoa(String.fromCharCode(...new Uint8Array(digest)));
      if (actual !== integrity) {
        throw new Error(`ファイルの整合性検証に失敗しました（期待: ${integrity} / 実際: ${actual}）`);
      }
    }

    async function cachedFetch(url, onProgress, integrity = '', signal = null) {
      // Only integrity-verified remote downloads are cached. Vendored local
      // assets are verified by tools/fetch-assets.py at build time; caching
      // them without a content hash could serve stale files after an update.
      const assertActive = () => {
        if (signal && signal.aborted) throw new DOMException('Piper fetch aborted', 'AbortError');
      };
      assertActive();
      let cache = null;
      if (integrity) {
        try {
          if (window.caches) cache = await caches.open(PIPER_CACHE_NAME);
        } catch (e) {
          console.warn('Cache API unavailable:', e);
        }
      }
      assertActive();
      const cacheKey = integrity ? `${url}?sarutahico_sri=${encodeURIComponent(integrity)}` : '';
      if (cache) {
        try {
          const hit = await cache.match(cacheKey);
          if (hit) {
            const cachedBlob = await hit.blob();
            await verifyBlobIntegrity(cachedBlob, integrity);
            assertActive();
            return cachedBlob;
          }
        } catch (e) {
          if (signal && signal.aborted) throw e;
          // A tampered or unreadable cache entry must not be trusted. Drop it
          // and fall through to a fresh, integrity-verified download.
          console.warn('Cached Piper asset rejected:', e);
          try { await cache.delete(cacheKey); } catch (deleteErr) { /* ignore */ }
        }
      }

      assertActive();
      const res = await fetch(url, signal ? { signal } : undefined);
      if (!res.ok) throw new Error(`${url} -> ${res.status}`);
      const total = Number(res.headers.get('Content-Length') || 0);
      let blob;
      if (res.body && typeof res.body.getReader === 'function' && total > 0 && onProgress) {
        const reader = res.body.getReader();
        const chunks = [];
        let loaded = 0;
        for (;;) {
          const { done, value } = await reader.read();
          if (done) break;
          chunks.push(value);
          loaded += value.length;
          onProgress(loaded, total);
        }
        blob = new Blob(chunks, { type: res.headers.get('Content-Type') || '' });
      } else {
        blob = await res.blob();
      }
      assertActive();
      if (integrity) await verifyBlobIntegrity(blob, integrity);
      if (cache) {
        try {
          await cache.put(cacheKey, new Response(blob, {
            headers: { 'Content-Type': blob.type || 'application/octet-stream' }
          }));
        } catch (e) {
          console.warn('Cache write failed:', e);
        }
      }
      assertActive();
      return blob;
    }

    async function pruneOldPiperCaches() {
      if (!window.caches || !caches.keys) return;
      try {
        const keys = await caches.keys();
        await Promise.all(keys
          .filter(key => key.startsWith('sarutahico-piper-') && key !== PIPER_CACHE_NAME)
          .map(key => caches.delete(key)));
      } catch (e) {
        console.warn('Old cache cleanup failed:', e);
      }
    }

    function pcmToWav(pcm, sampleRate) {
      const buffer = new ArrayBuffer(44 + pcm.length * 2);
      const view = new DataView(buffer);
      const writeText = (offset, text) => {
        for (let i = 0; i < text.length; i++) view.setUint8(offset + i, text.charCodeAt(i));
      };
      writeText(0, 'RIFF');
      view.setUint32(4, buffer.byteLength - 8, true);
      writeText(8, 'WAVE');
      writeText(12, 'fmt ');
      view.setUint32(16, 16, true);
      view.setUint16(20, 1, true);
      view.setUint16(22, 1, true);
      view.setUint32(24, sampleRate, true);
      view.setUint32(28, sampleRate * 2, true);
      view.setUint16(32, 2, true);
      view.setUint16(34, 16, true);
      writeText(36, 'data');
      view.setUint32(40, pcm.length * 2, true);
      for (let i = 0; i < pcm.length; i++) {
        const x = Math.max(-1, Math.min(1, pcm[i]));
        view.setInt16(44 + i * 2, x < 0 ? x * 0x8000 : x * 0x7fff, true);
      }
      return buffer;
    }

    async function getPiperSession(voiceId) {
      const voice = PIPER_VOICES[voiceId];
      if (!voice) throw new Error('Unknown Piper voice: ' + voiceId);
      if (piperSession && piperSession.voiceId === voiceId) return piperSession;
      if (piperLoading && piperLoading.voiceId === voiceId) return piperLoading.promise;

      // A preparation promise can be joined by a new request after a stop.
      // Only show its progress while this voice is still the active request.
      // Remote model downloads get both a deadline and an abort handle so a
      // stalled fallback cannot keep playback preparation pending forever.
      const fetchController = new AbortController();
      const fetchTimeoutId = setTimeout(() => fetchController.abort(), PIPER_FETCH_TIMEOUT_MS);
      piperFetchControllers.add(fetchController);
      const setPrepStatus = (text) => {
        if (!(activeButton && voiceSelect.value === voiceId)) return;
        const match = text.match(/(\d+)%/);
        if (match) {
          const pct = Number(match[1]);
          if (pct !== 0 && pct !== 100 && pct % 10 !== 0) return;
        }
        setTtsStatus(text);
      };
      const promise = (async () => {
        await loadPiperRuntime();
        const modelAsset = await resolveAsset(
          `vendor/piper/models/${voice.path}.onnx`,
          `${PIPER_VOICES_URL}/${voice.path}.onnx`,
          { requireSri: false } // PIPER_ASSET_SRI でダウンロード後に検証する
        );
        const base = modelAsset.local ? `vendor/piper/models/${voice.path}` : `${PIPER_VOICES_URL}/${voice.path}`;
        // Vendored files are verified by tools/fetch-assets.py. A remote
        // fallback is unverified, so check its SHA-256 before use/cache.
        const configSri = modelAsset.local ? '' : (PIPER_ASSET_SRI[voice.path + '.onnx.json'] || '');
        const modelSri = modelAsset.local ? '' : (PIPER_ASSET_SRI[voice.path + '.onnx'] || '');
        if (!modelAsset.local && (!configSri || !modelSri)) {
          throw new Error('この音声の SRI が定義されていません。');
        }
        setPrepStatus('ニューラル音声の設定を読み込み中...');
        const configBlob = await cachedFetch(base + '.onnx.json', null, configSri, fetchController.signal);
        const config = JSON.parse(await configBlob.text());
        setPrepStatus('ニューラル音声のモデルを読み込み中...');
        const modelBlob = await cachedFetch(base + '.onnx', (loaded, total) => {
          const pct = Math.round(loaded * 100 / total);
          setPrepStatus(`モデルを取得中... ${pct}%`);
        }, modelSri, fetchController.signal);
        const session = await window.ort.InferenceSession.create(await modelBlob.arrayBuffer(), {
          executionProviders: ['wasm']
        });

        // Follow piper1-gpl's pinyin frontend so the ID sequence matches training:
        // zero initial "Ø", ü-family "v", neutral tone 5, and punctuation symbols.
        // This avoids the espeak phonemizer and its 18 MB assets.
        const idMap = config.phoneme_id_map || {};
        const has = (symbol) => Array.isArray(idMap[symbol]);
        const phonemize = (text) => {
          // Digits are not in the phoneme map. Expand them before pinyin
          // conversion; years are read digit-by-digit with 〇.
          const source = expandChineseNumbers(text).replace(/\s+/g, ' ');
          const syllables = window.pinyinPro.pinyin(source, {
            toneType: 'num', type: 'array', nonZh: 'consecutive'
          }).map(item => String(item));
          const phonemes = [];
          const dropped = [];
          syllables.forEach(item => {
            const symbol = String(item);
            const match = symbol.toLowerCase().match(/^([a-zü]+)([0-5])$/);
            if (!match) {
              // pinyin-pro groups consecutive non-Chinese characters, e.g.
              // "。 " or "！！"; the voice map only has single symbols.
              // Keep every known character in order instead of dropping the
              // whole cluster.
              Array.from(symbol).forEach(single => {
                // Tone-less pinyin-like input can reach here (e.g. "nü").
                // The voice map uses "v" for ü; other characters are kept
                // as-is so arbitrary Latin text is not mispronounced.
                const normalized = single === 'ü' ? 'v' : single;
                if (has(normalized)) phonemes.push(normalized);
                else dropped.push(single);
              });
              return;
            }
            let body = match[1].replace(/ü/g, 'v');
            const initial = PINYIN_INITIALS.find(x => body.startsWith(x) && has(x));
            if (initial) {
              phonemes.push(initial);
              body = body.slice(initial.length);
              // pinyin-pro already matches upstream g2pW for j/q/x/y finals
              // (ju/quan/yue/yun stay plain u; see upstream test_phonemize:
              // 语 -> ["y","u","3"]). Only literal ü from n/l was normalized
              // to v above. Do not add an extra u->v conversion here: it
              // would deviate from the sequence used to train chaowen.
            } else if (has('Ø')) {
              phonemes.push('Ø');
            }
            if (body && has(body)) phonemes.push(body);
            else if (body) dropped.push(body);
            // pinyin-pro emits tone 0 for neutral syllables; chaowen's map uses 5.
            const tone = match[2] === '0' ? '5' : match[2];
            if (tone && has(tone)) phonemes.push(tone);
          });
          if (dropped.length) console.warn('[piper] Unknown pinyin symbols omitted:', dropped);
          return { phonemes };
        };
        piperSession = { voiceId, session, config, phonemize };
        setPrepStatus(`Piper（${voice.label}）の準備ができました`);
        return piperSession;
      })();

      piperLoading = { voiceId, promise };
      try {
        return await promise;
      } finally {
        clearTimeout(fetchTimeoutId);
        piperFetchControllers.delete(fetchController);
        if (piperLoading && piperLoading.promise === promise) piperLoading = null;
      }
    }

    async function speakWithPiper(text, btn, token) {
      if (!beginPlaybackUI(btn, token)) return;
      const requestedVoiceId = voiceSelect.value;
      let audioErrorHandled = false;
      // One fallback path for every Piper failure. audio.onerror and a
      // subsequent audio.play() rejection can both fire for the same media
      // error; only the first one may switch voices and notify the user.
      const fallbackToBrowser = (reason = 'prepare') => {
        if (token !== playbackToken) return;
        if (activeAudio) {
          const audio = activeAudio;
          activeAudio = null;
          audio.onended = null;
          audio.onerror = null;
          try { audio.pause(); audio.currentTime = 0; } catch (e) { /* ignore */ }
        }
        if (activeAudioUrl) {
          URL.revokeObjectURL(activeAudioUrl);
          activeAudioUrl = null;
        }
        resetPlaybackUI();
        const fallbackValue = browserVoiceDefault() || '';
        voiceSelect.value = fallbackValue;
        fallbackVoiceRestore = { restore: requestedVoiceId, temporary: fallbackValue };
        updateSelectedVoice();
        updateTtsControls();
        const message = reason === 'audio'
          ? '音声の再生に失敗したため、ブラウザ音声で再生します。'
          : reason === 'noPhonemes'
            ? 'この文字は中国語音声で読み上げられません。ブラウザ音声で再生します。'
            : 'ニューラル音声の準備に失敗したため、ブラウザ音声で再生します。';
        showMessage(message, 5000);
        speakWithBrowser(text, btn, token, 'ブラウザ音声に切り替えて再生中', () => {
          // The fallback is for this playback only. Stop/replacement also
          // restores the voice, so this is just the normal completion path.
          restoreFallbackVoice();
          // restoreFallbackVoice() updates controls and can reselect the Piper
          // status; playback has ended, so clear it like every other path.
          setTtsStatus('');
        });
      };
      try {
        const voiceId = voiceSelect.value;
        const engine = await getPiperSession(voiceId);
        if (token !== playbackToken) return;
        setTtsStatus('ニューラル音声を生成中...');
        const phonemes = engine.phonemize(text);
        if (token !== playbackToken) return;
        // Map by symbol so IDs outside the model's embedding cannot reach it.
        // piper1-gpl resets BOS/EOS and runs inference per sentence.
        const idMap = engine.config.phoneme_id_map || {};
        const idOf = (symbol, fallback) => Array.isArray(idMap[symbol]) ? Number(idMap[symbol][0]) : fallback;
        const bos = idOf('^', 1);
        const eos = idOf('$', 2);
        const pad = idOf('_', 0);
        const sentenceEnd = new Set(['。', '？', '?', '！', '!', '…', '.']);
        const dropped = [];
        const groups = [];
        let current = [];
        let pendingSentenceEnd = false;
        // Upstream stream_to_sentences() treats a consecutive run of sentence
        // enders as one boundary. Splitting at every symbol would add extra
        // BOS/EOS/pad pairs for input such as 你好！！！.
        (phonemes.phonemes || []).forEach(phoneme => {
          const isSentenceEnd = sentenceEnd.has(phoneme);
          if (pendingSentenceEnd && !isSentenceEnd) {
            groups.push(current);
            current = [];
            pendingSentenceEnd = false;
          }
          current.push(phoneme);
          if (isSentenceEnd) pendingSentenceEnd = true;
        });
        if (current.length) groups.push(current);

        const idGroups = [];
        groups.forEach(group => {
          if (!group.some(symbol => symbol !== ' ')) return;
          const ids = [bos];
          group.forEach(phoneme => {
            const mapped = idMap[phoneme];
            if (!Array.isArray(mapped)) {
              dropped.push(phoneme);
              return;
            }
            mapped.forEach(id => ids.push(Number(id)));
            if (PIPER_GROUP_END.has(phoneme)) ids.push(pad);
          });
          ids.push(eos);
          if (ids.length > 3) idGroups.push(ids);
        });
        if (dropped.length) console.warn('[piper] Unsupported phonemes omitted:', dropped);
        if (token !== playbackToken) return;
        if (!idGroups.length) throw new Error('No phonemes recognized by the voice model');

        const inf = engine.config.inference;
        const scales = Float32Array.from([
          inf.noise_scale, inf.length_scale / ttsRate, inf.noise_w
        ]);
        const speakerIds = engine.config.speaker_id_map && Object.keys(engine.config.speaker_id_map).length
          ? BigInt64Array.from([0n])
          : null;
        const pcmChunks = [];
        let pcmLength = 0;
        for (const ids of idGroups) {
          const feeds = {
            input: new window.ort.Tensor('int64', BigInt64Array.from(ids, BigInt), [1, ids.length]),
            input_lengths: new window.ort.Tensor('int64', BigInt64Array.from([BigInt(ids.length)])),
            scales: new window.ort.Tensor('float32', Float32Array.from(scales))
          };
          if (speakerIds) feeds.sid = new window.ort.Tensor('int64', speakerIds);
          const result = await engine.session.run(feeds);
          if (token !== playbackToken) return;
          const outputTensor = result.output || result[Object.keys(result)[0]];
          const chunk = new Float32Array(outputTensor.data.length);
          chunk.set(outputTensor.data);
          pcmChunks.push(chunk);
          pcmLength += chunk.length;
        }
        const pcm = new Float32Array(pcmLength);
        let pcmOffset = 0;
        pcmChunks.forEach(chunk => {
          pcm.set(chunk, pcmOffset);
          pcmOffset += chunk.length;
        });
        const wav = pcmToWav(pcm, engine.config.audio.sample_rate);
        const url = URL.createObjectURL(new Blob([wav], { type: 'audio/wav' }));
        if (token !== playbackToken || activeButton !== btn) {
          URL.revokeObjectURL(url);
          return;
        }
        if (activeAudioUrl) URL.revokeObjectURL(activeAudioUrl);
        const audio = new Audio(url);
        activeAudioUrl = url;
        activeAudio = audio;
        activeAudio.dataset.engine = 'piper';
        audio.onended = () => {
          if (activeAudio !== audio) return;
          activeAudio = null;
          if (activeAudioUrl === url) {
            URL.revokeObjectURL(url);
            activeAudioUrl = null;
          }
          endPlaybackUI(token);
          setTtsStatus('');
        };
        audio.onerror = (e) => {
          if (activeAudio !== audio) return;
          audioErrorHandled = true;
          console.error('Piper audio error:', e);
          activeAudio = null;
          if (activeAudioUrl === url) {
            URL.revokeObjectURL(url);
            activeAudioUrl = null;
          }
          fallbackToBrowser('audio');
        };
        await audio.play();
        if (token !== playbackToken || activeAudio !== audio) return;
        setTtsStatus(`Piper（${PIPER_VOICES[voiceId].label}）で再生中`);
      } catch (err) {
        // audio.onerror already started the single fallback; the rejected
        // play() promise must not switch voices or notify a second time.
        if (audioErrorHandled) return;
        if (token !== playbackToken) return;
        console.error('Piper TTS error:', err);
        const noPhonemes = err && err.message === 'No phonemes recognized by the voice model';
        fallbackToBrowser(noPhonemes ? 'noPhonemes' : 'prepare');
      }
    }

    function playText(text, btn) {
      if (!text) return;
      const sameButton = activeButton === btn;
      const speaking = window.speechSynthesis && window.speechSynthesis.speaking;
      if (activeButton || activeAudio || activeUtterance || speaking) {
        stopSpeaking();
        if (sameButton) return;
      }
      const token = ++playbackToken;
      if (voiceSelect.value.startsWith('piper:')) {
        speakWithPiper(text, btn, token);
      } else {
        speakWithBrowser(text, btn, token);
      }
    }

    playPinyinBtn.addEventListener('click', () => playText(currentKanjiText, playPinyinBtn));
    playExampleBtn.addEventListener('click', () => playText(currentExampleText, playExampleBtn));

    // ---- 手動辞書検索 ----
    function displayResult(result) {
      stopSpeaking();
      detectedKanjiEl.textContent = result.kanji || '--';
      meaningEl.textContent = result.meaning || '--';
      pinyinEl.textContent = result.pinyin || '--';
      exampleEl.textContent = result.example || '--';
      currentKanjiText = (result.kanji && result.kanji !== '--') ? result.kanji.charAt(0) : '';
      let exampleChinese = result.example || '';
      if (exampleChinese.includes('/')) {
        exampleChinese = exampleChinese.split('/')[0].trim();
      }
      const unavailableExample = /^（[^）]*未登録[^）]*）$/.test(exampleChinese);
      currentExampleText = (exampleChinese && exampleChinese !== '--' && !unavailableExample) ? exampleChinese : '';
      playPinyinBtn.disabled = !currentKanjiText;
      playExampleBtn.disabled = !currentExampleText;
    }

    lookupBtn.addEventListener('click', () => {
      cancelAnalysis();
      const kanji = manualKanjiInput.value.trim();
      if (!kanji) {
        showMessage('漢字を1文字入力してください。');
        return;
      }
      const result = lookupDictionary(kanji.charAt(0));
      displayResult(result);
      if (result.meaning && result.meaning.includes('未登録')) {
        showMessage('内蔵辞書に未登録です。API連携モードを使うか別の文字を試してください。', 5000);
      }
    });

    manualKanjiInput.addEventListener('keydown', (e) => {
      // Enter commits an IME candidate; do not search the uncommitted value.
      if (e.key === 'Enter' && !e.isComposing && e.keyCode !== 229) {
        lookupBtn.click();
      }
    });

    // ---- メイン処理 ----
    analyzeBtn.addEventListener('click', async () => {
      if (analysisActive || analyzeBtn.getAttribute('aria-disabled') === 'true') return;
      if (isCanvasEmpty()) {
        showMessage('キャンバスが空です。文字を描くか画像をアップロードしてください。');
        return;
      }
      const focusBeforeAnalysis = document.activeElement;
      const generation = ++analysisGeneration;
      analysisActive = true;
      let completed = false;
      try {
        analyzeBtn.setAttribute('aria-disabled', 'true');
        analyzeBtn.innerHTML = '<span class="spinner"></span>解析中...';
        resultPanel.setAttribute('aria-busy', 'true');
        analysisStatusEl.textContent = '解析中です';
        stopSpeaking();

        detectedKanjiEl.textContent = '--';
        meaningEl.textContent = '--';
        pinyinEl.textContent = '--';
        exampleEl.textContent = '--';
        currentKanjiText = '';
        currentExampleText = '';
        playPinyinBtn.disabled = true;
        playExampleBtn.disabled = true;

        const dataUrl = canvas.toDataURL('image/png');

        if (useApiMode) {
          // APIモード
          const kanji = await extractKanjiFromImage(dataUrl);
          if (generation !== analysisGeneration) return;
          if (kanji === null) {
            meaningEl.textContent = lastMessageText || 'APIの設定を確認してください。';
            return;
          }
          if (!kanji) {
            meaningEl.textContent = '漢字が見つかりませんでした。はっきりとした文字を描くか、別の画像を試してください。';
            return;
          }
          const result = await generateDetailsForKanji(kanji);
          if (generation !== analysisGeneration) return;
          if (!result) return;
          displayResult(result);
          completed = true;
        } else {
          // オフラインモード: 画像前処理 + Tesseract.js OCR
          try {
            showProgress(true);
            setProgress(5);
            showMessage('OCR認識中...（初回は言語データの読み込みに時間がかかります）', 15000);

            // Preprocess image for better OCR
            const processedDataUrl = preprocessCanvas();
            setProgress(15);

            const kanji = await runOCR(processedDataUrl, { manageProgress: false });
            if (generation !== analysisGeneration) return;
            setProgress(90);

            if (kanji) {
              const result = lookupDictionary(kanji);
              displayResult(result);
              completed = true;
              if (result.meaning === '（辞書に未登録）') {
                showMessage('認識された漢字: ' + kanji + '（内蔵辞書に未登録です。API連携モードを使うか別の文字を試してください）', 5000);
              } else {
                showMessage('認識された漢字: ' + kanji, 3000);
              }
            } else {
              // Fallback: try original image without preprocessing
              const kanji2 = await runOCR(dataUrl, { manageProgress: false });
              if (generation !== analysisGeneration) return;
              if (kanji2) {
                const result = lookupDictionary(kanji2);
                displayResult(result);
                completed = true;
                showMessage('認識された漢字: ' + kanji2, 3000);
              } else {
                showMessage('漢字を認識できませんでした。手動入力をお使いください。', 5000);
                meaningEl.textContent = 'API不要モード: 下の入力欄に漢字を入力して検索してください。';
              }
            }
            setProgress(100);
            scheduleProgressHide(1000);
          } catch (ocrErr) {
            if (generation !== analysisGeneration) return;
            console.error('OCR Error:', ocrErr && ocrErr.message ? ocrErr.message : ocrErr);
            showProgress(false);
            showMessage('OCR認識に失敗しました。手動入力をお使いください。' + (ocrErr && ocrErr.message ? ' ' + ocrErr.message : ''), 0);
            meaningEl.textContent = 'API不要モード: 下の入力欄に漢字を入力して検索してください。';
          }
        }

      } catch (err) {
        if (generation !== analysisGeneration) return;
        console.error('Analysis Error:', err);
        showMessage('解析中に予期せぬエラーが発生しました。');
      } finally {
        if (generation !== analysisGeneration) return;
        analysisActive = false;
        analyzeBtn.removeAttribute('aria-disabled');
        analyzeBtn.textContent = '漢字分析';
        resultPanel.removeAttribute('aria-busy');
        analysisStatusEl.textContent = completed ? '解析が完了しました' : '解析に失敗しました';
        const currentFocus = document.activeElement;
        if (focusBeforeAnalysis && focusBeforeAnalysis.isConnected &&
            typeof focusBeforeAnalysis.focus === 'function' &&
            (currentFocus === focusBeforeAnalysis || currentFocus === document.body)) {
          focusBeforeAnalysis.focus();
        }
      }
    });

  </script>
</body>
</html>
